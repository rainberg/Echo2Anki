```python
from uuid import UUID

from fastapi import APIRouter, Depends, HTTPException, status
from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession

from app.database import get_db
from app.deps import get_admin_user
from app.models import HelpArticle, User
from app.schemas_help import (
    HelpArticleCreate,
    HelpArticlePublicResponse,
    HelpArticleResponse,
    HelpArticleSummary,
    HelpArticleUpdate,
    HelpCategoryResponse,
)

CATEGORY_LABELS = {
    "tasks": "任务与音频处理",
    "collections": "收藏与公共库",
    "anki": "Anki 推送",
}

router = APIRouter(prefix="/help", tags=["help"])
admin_help_router = APIRouter(prefix="/admin/help", tags=["admin-help"])


# ─── Public endpoints ──────────────────────────────────────────────────


@router.get("/categories", response_model=list[HelpCategoryResponse])
async def list_help_categories(db: AsyncSession = Depends(get_db)):
    result = await db.execute(
        select(HelpArticle)
        .where(HelpArticle.is_published == True)
        .order_by(HelpArticle.category, HelpArticle.sort_order, HelpArticle.title)
    )
    articles = result.scalars().all()

    grouped: dict[str, list[HelpArticleSummary]] = {}
    for a in articles:
        grouped.setdefault(a.category, []).append(HelpArticleSummary(slug=a.slug, title=a.title))

    response = []
    for cat_key, label in CATEGORY_LABELS.items():
        if cat_key in grouped:
            response.append(HelpCategoryResponse(category=cat_key, label=label, articles=grouped[cat_key]))
    return response


@router.get("/articles/{slug}", response_model=HelpArticlePublicResponse)
async def get_help_article(slug: str, db: AsyncSession = Depends(get_db)):
    result = await db.execute(
        select(HelpArticle).where(HelpArticle.slug == slug, HelpArticle.is_published == True)
    )
    article = result.scalar_one_or_none()
    if not article:
        raise HTTPException(status_code=status.HTTP_404_NOT_FOUND, detail="Article not found")
    return article


# ─── Admin endpoints ───────────────────────────────────────────────────


@admin_help_router.get("/articles", response_model=list[HelpArticleResponse])
async def admin_list_articles(
    category: str = None,
    admin: User = Depends(get_admin_user),
    db: AsyncSession = Depends(get_db),
):
    query = select(HelpArticle).order_by(HelpArticle.category, HelpArticle.sort_order, HelpArticle.title)
    if category:
        query = query.where(HelpArticle.category == category)
    result = await db.execute(query)
    return result.scalars().all()


@admin_help_router.post("/articles", response_model=HelpArticleResponse, status_code=status.HTTP_201_CREATED)
async def admin_create_article(
    body: HelpArticleCreate,
    admin: User = Depends(get_admin_user),
    db: AsyncSession = Depends(get_db),
):
    existing = await db.execute(select(HelpArticle).where(HelpArticle.slug == body.slug))
    if existing.scalar_one_or_none():
        raise HTTPException(status_code=status.HTTP_409_CONFLICT, detail="Slug already exists")
    article = HelpArticle(**body.model_dump())
    db.add(article)
    await db.commit()
    await db.refresh(article)
    return article


@admin_help_router.put("/articles/{article_id}", response_model=HelpArticleResponse)
async def admin_update_article(
    article_id: UUID,
    body: HelpArticleUpdate,
    admin: User = Depends(get_admin_user),
    db: AsyncSession = Depends(get_db),
):
    result = await db.execute(select(HelpArticle).where(HelpArticle.id == article_id))
    article = result.scalar_one_or_none()
    if not article:
        raise HTTPException(status_code=status.HTTP_404_NOT_FOUND, detail="Article not found")

    if body.slug is not None and body.slug != article.slug:
        existing = await db.execute(select(HelpArticle).where(HelpArticle.slug == body.slug))
        if existing.scalar_one_or_none():
            raise HTTPException(status_code=status.HTTP_409_CONFLICT, detail="Slug already exists")

    update_data = body.model_dump(exclude_unset=True)
    for key, value in update_data.items():
        setattr(article, key, value)
    await db.commit()
    await db.refresh(article)
    return article


@admin_help_router.delete("/articles/{article_id}", status_code=status.HTTP_204_NO_CONTENT)
async def admin_delete_article(
    article_id: UUID,
    admin: User = Depends(get_admin_user),
    db: AsyncSession = Depends(get_db),
):
    result = await db.execute(select(HelpArticle).where(HelpArticle.id == article_id))
    article = result.scalar_one_or_none()
    if not article:
        raise HTTPException(status_code=status.HTTP_404_NOT_FOUND, detail="Article not found")
    await db.delete(article)
    await db.commit()
```

import { ref, computed, onMounted, watch } from 'vue'
import { useRoute, useRouter } from 'vue-router'
import { useMessage } from 'naive-ui'
import { marked } from 'marked'
import { getHelpCategories, getHelpArticle } from '../api/help'

const route = useRoute()
const router = useRouter()
const message = useMessage()

const categories = ref([])
const article = ref(null)
const currentSlug = ref(null)
const loading = ref(false)

const renderedContent = computed(() => {
  if (!article.value?.content) return ''
  return marked(article.value.content)
})

async function fetchCategories() {
  loading.value = true
  try {
    const { data } = await getHelpCategories()
    categories.value = data
  } catch {
    message.error('获取帮助分类失败')
  } finally {
    loading.value = false
  }
}

async function loadArticle(slug) {
  currentSlug.value = slug
  router.replace(`/help/${slug}`)
  try {
    const { data } = await getHelpArticle(slug)
    article.value = data
  } catch {
    message.error('获取文章失败')
    article.value = null
  }
}

onMounted(async () => {
  await fetchCategories()
  // If URL has a slug, load it; otherwise load first article
  const slug = route.params.slug
  if (slug) {
    await loadArticle(slug)
  } else if (categories.value.length > 0 && categories.value[0].articles.length > 0) {
    await loadArticle(categories.value[0].articles[0].slug)
  }
})

watch(() => route.params.slug, (newSlug) => {
  if (newSlug && newSlug !== currentSlug.value) {
    loadArticle(newSlug)
  }
})

async function fetchHelpArticles() {
  helpLoading.value = true
  try {
    const params = {}
    if (helpCategoryFilter.value) params.category = helpCategoryFilter.value
    const { data } = await adminListHelpArticles(params)
    helpArticles.value = data
  } catch {
    message.error('获取帮助文章失败')
  } finally {
    helpLoading.value = false
  }
}

function openHelpCreate() {
  helpEditIsNew.value = true
  helpEdit.value = { id: null, slug: '', title: '', content: '', category: 'tasks', sort_order: 0, is_published: false }
  helpEditVisible.value = true
}

function openHelpEdit(article) {
  helpEditIsNew.value = false
  helpEdit.value = { ...article }
  helpEditVisible.value = true
}

async function handleSaveHelpArticle() {
  try {
    if (helpEditIsNew.value) {
      await adminCreateHelpArticle({
        slug: helpEdit.value.slug,
        title: helpEdit.value.title,
        content: helpEdit.value.content,
        category: helpEdit.value.category,
        sort_order: helpEdit.value.sort_order,
        is_published: helpEdit.value.is_published,
      })
      message.success('文章已创建')
    } else {
      await adminUpdateHelpArticle(helpEdit.value.id, {
        title: helpEdit.value.title,
        content: helpEdit.value.content,
        category: helpEdit.value.category,
        sort_order: helpEdit.value.sort_order,
        is_published: helpEdit.value.is_published,
      })
      message.success('文章已更新')
    }
    fetchHelpArticles()
  } catch (e) {
    const detail = e.response?.data?.detail || '操作失败'
    message.error(detail)
  }
}

async function handleDeleteHelpArticle(article) {
  if (!confirm(`确定删除「${article.title}」？`)) return
  try {
    await adminDeleteHelpArticle(article.id)
    message.success('已删除')
    fetchHelpArticles()
  } catch {
    message.error('删除失败')
  }
}
